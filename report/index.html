// WatchBack tester report — the data half. // // This returns JSON, not a page. Supabase rewrites every Edge Function response // on the default *.supabase.co domain to `Content-Type: text/plain` with a // sandbox CSP, deliberately, so the functions domain cannot be used to serve // web content. A browser therefore shows the source of any HTML sent from here // instead of rendering it. The page that consumes this lives on GitHub Pages, // which serves real HTML for free. // // Threat model, since this URL is public and the data behind it is tester email // addresses: // // * The passkey exists in no system at all — only sha256(salt || passkey) is // stored — so this file, this repo and the database all leak nothing. // * It is compared as a digest, not as a string, so the time taken to reject // a wrong key says nothing about how wrong it was. // * It is only ever sent in a POST body. A key in the query string ends up in // browser history, in server logs, and in the Referer of every outbound // link. // * Five wrong tries locks that caller out for 15 minutes, counted in the // database because an Edge Function remembers nothing between requests. The // lockout is keyed on a salted hash of the IP, so this table never becomes // a record of where the owner has been. // * A correct key returns a signed 8-hour token. The page holds it in // sessionStorage, so closing the tab throws it away. // * CORS names one origin. A page on any other site cannot read a reply, even // holding a valid token. // * Not one byte of tester data is read from the database before the token or // the passkey checks out. // // Rotate the passkey with: select admin.set_report_passkey('NEW-KEY'); import { createClient } from 'jsr:@supabase/supabase-js@2'; const SESSION_HOURS = 8; // The one page allowed to read a reply. Anything else gets no CORS header and // so cannot see the body, whatever it sends. const ALLOWED_ORIGINS = [ 'https://infinix10prozeestarzone-rgb.github.io', 'http://localhost:8000', ]; const enc = new TextEncoder(); interface Secrets { salt: string; passkey_sha256: string; session_secret: string; } function toHex(buf: ArrayBuffer): string { return [...new Uint8Array(buf)].map((b) => b.toString(16).padStart(2, '0')).join(''); } async function sha256(text: string): Promise { return toHex(await crypto.subtle.digest('SHA-256', enc.encode(text))); } async function hmac(secret: string, message: string): Promise { const key = await crypto.subtle.importKey( 'raw', enc.encode(secret), { name: 'HMAC', hash: 'SHA-256' }, false, ['sign'], ); return toHex(await crypto.subtle.sign('HMAC', key, enc.encode(message))); } /** * Compares two equal-length hex digests without leaking where they differ. * * Every caller passes digests, never raw input, so the loop always runs over 64 * characters and its duration says nothing about how close a guess was. */ function sameDigest(a: string, b: string): boolean { if (a.length !== b.length) return false; let diff = 0; for (let i = 0; i < a.length; i++) diff |= a.charCodeAt(i) ^ b.charCodeAt(i); return diff === 0; } async function issueToken(sec: Secrets): Promise { const exp = Date.now() + SESSION_HOURS * 3600_000; return `${exp}.${await hmac(sec.session_secret, String(exp))}`; } async function tokenValid(token: string, sec: Secrets): Promise { const dot = token.lastIndexOf('.'); if (dot < 1) return false; const exp = token.slice(0, dot); const sig = token.slice(dot + 1); // Expiry first, and only via a digits-only regex, so a hostile value never // reaches Number(). The signature covers the expiry, so pushing the date out // invalidates the token rather than extending it. if (!/^\d+$/.test(exp) || Number(exp) < Date.now()) return false; return sameDigest(sig, await hmac(sec.session_secret, exp)); } /** Salted hash of the caller's IP, so the rate-limit table holds no addresses. */ async function ipHash(req: Request, sec: Secrets): Promise { const ip = (req.headers.get('x-forwarded-for') ?? '').split(',')[0].trim() || 'unknown'; return await hmac(sec.session_secret, `ip:${ip}`); } function corsFor(req: Request): Record { const origin = req.headers.get('origin') ?? ''; if (!ALLOWED_ORIGINS.includes(origin)) return {}; return { 'access-control-allow-origin': origin, 'access-control-allow-headers': 'content-type', 'access-control-allow-methods': 'POST, OPTIONS', 'vary': 'Origin', }; } function json(body: unknown, status: number, req: Request) { return new Response(JSON.stringify(body), { status, headers: { 'content-type': 'application/json; charset=utf-8', 'cache-control': 'no-store', 'x-content-type-options': 'nosniff', ...corsFor(req), }, }); } Deno.serve(async (req: Request) => { if (req.method === 'OPTIONS') { return new Response(null, { status: 204, headers: corsFor(req) }); } if (req.method !== 'POST') { return json({ error: 'post_only' }, 405, req); } const url = new URL(req.url); // A passkey in the query string would sit in browser history and in every log // along the way. Refuse it rather than quietly accepting it. if (url.searchParams.has('key') || url.searchParams.has('passkey')) { return json({ error: 'no_key_in_url' }, 400, req); } const admin = createClient( Deno.env.get('SUPABASE_URL')!, Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!, { auth: { persistSession: false } }, ); // Everything goes through public.report_* wrappers: the admin schema is not // exposed to PostgREST, and it should stay that way. const { data: secRows } = await admin.rpc('report_secrets'); const sec = (secRows ?? [])[0] as Secrets | undefined; if (!sec) return json({ error: 'not_configured' }, 503, req); let payload: { key?: string; token?: string }; try { payload = await req.json(); } catch { return json({ error: 'bad_request' }, 400, req); } const hash = await ipHash(req, sec); let token = typeof payload.token === 'string' ? payload.token : ''; let authed = token ? await tokenValid(token, sec) : false; if (!authed) { const { data: lock } = await admin .rpc('report_gate', { p_ip_hash: hash, p_action: 'check' }); if (Number(lock ?? 0) > 0) { return json({ error: 'locked', seconds: Number(lock) }, 429, req); } const submitted = String(payload.key ?? '').trim().toUpperCase(); const digest = await sha256(sec.salt + submitted); if (!sameDigest(digest, sec.passkey_sha256)) { const { data: locked } = await admin .rpc('report_gate', { p_ip_hash: hash, p_action: 'fail' }); // One reply for every failure. Distinguishing "wrong passkey" from // "locked out" tells an attacker which guesses were worth making. return json( { error: 'denied', seconds: Number(locked ?? 0) }, Number(locked ?? 0) > 0 ? 429 : 401, req, ); } await admin.rpc('report_gate', { p_ip_hash: hash, p_action: 'ok' }); token = await issueToken(sec); authed = true; } // ---- authenticated; only now is any tester data touched ---- // Added .limit(10000) to override Supabase's default 1000-row limit API restriction const { data: pd, error: pdErr } = await admin.rpc('report_person_day').limit(10000); const { data: daily } = await admin.rpc('report_daily').limit(10000); // What each person did and when, so a disputed wallet screenshot can be // settled without opening the database. const { data: acts } = await admin.rpc('report_person_actions', { p_email: null, p_days: 3, }).limit(10000); // Opened the app and did nothing with it. Kept apart from the work counts // rather than folded into them: the grid used to render an open as a 1, which // made a person who only looked read the same as one who watched a video. const { data: opens } = await admin.rpc('report_person_opens').limit(10000); if (pdErr) return json({ error: 'read_failed' }, 500, req); return json({ token, expires_in: SESSION_HOURS * 3600, // generated_at stays a parseable ISO string. The deployed page runs // new Date() on it, and swapping in a pre-formatted string turned that into // "Invalid Date" -- a served page is not redeployed in lockstep with this // function, so never change the shape of a field it already reads. generated_at: new Date().toISOString(), // The pre-formatted one is additive: a new page prefers it, an old page // ignores it and keeps working. generated_at_pkt: new Date().toLocaleString('en-GB', { timeZone: 'Asia/Karachi', dateStyle: 'medium', timeStyle: 'short', }) + ' PKT', person_day: pd ?? [], daily: daily ?? [], person_actions: acts ?? [], // Additive, like generated_at_pkt: a page that does not know about this // field ignores it and keeps working. person_opens: opens ?? [], }, 200, req); });
